\section{套路A：用药IPW（糖尿病 $\times$ 28天死亡）}

\subsection{在做什么}
\begin{enumerate}
  \item \textbf{定义暴露}：MICE后由 HbA1c$\geq$6.5\% 派生 \texttt{Diabetes\_HbA1c}；原始HbA1c与糖尿病诊断列硬排除，不进PS。
  \item \textbf{协变量}：单因素Cox（$P<0.10$）$\rightarrow$ VIF $\rightarrow$ 临床 \texttt{Model2Factors}（\textbf{不含}实验室复合指标、不含暴露）。
  \item \textbf{预后风险分（仅供STEPP）}：用Model2Factors拟合28天死亡Cox，得 \texttt{composite\_risk}=LP（对齐Jin；\textbf{不是}NLR类指标）。
  \item \textbf{IPW主分析}：PS加权后，比较糖尿病 vs 非糖尿病的28天全因死亡（加权KM + IPW-Cox）。
  \item \textbf{扩展}：亚组森林、年龄分层KM、STEPP（两臂=Diabetes，$\mathrm{x}$=预后风险分）、敏感性Cox/重叠权重、校准+ROC。
  \item \textbf{架构}：单次 \texttt{unit=main}；\textbf{无}实验室复合指标并行。
\end{enumerate}

\subsection{需要什么样的数据}
\begin{itemize}
  \item 队列：基线宽表 + 预后；暴露用HbA1c；结局用28天生存时间/死亡指示。
  \item 协变量：人口学、合并症、严重程度、常规实验室等（供筛选进PS）。
  \item \textbf{明确不需要}：NLR等实验室复合指标、指标日序列、出院竞争事件编码。
\end{itemize}

\subsection{典型流水线}
\begin{center}
\small
\texttt{clean $\rightarrow$ mapping $\rightarrow$ impute $\rightarrow$ 暴露派生 $\rightarrow$ 硬排除}\\
\texttt{$\rightarrow$ 单因素 $\rightarrow$ VIF $\rightarrow$ 预后风险分 $\rightarrow$ IPW}\\
\texttt{$\rightarrow$ KM/亚组/STEPP/敏感性/校准 $\rightarrow$ pub}
\end{center}

\section{套路B：竞争风险（实验室复合指标 $\times$ 28天糖尿病）}

\subsection{在做什么}
\begin{enumerate}
  \item 28天内编码：status=1糖尿病（首次HbA1c$\geq$6.5），2死亡，3出院，0删失。
  \item 暴露=当前\textbf{实验室复合指标}的四分位 + 28天轨迹分型。
  \item 主分析：CIF + Fine-Gray（Model~1--3）；死亡侧、RCS、轨迹图等。
  \item 特征：单因素$\rightarrow$LASSO$\rightarrow$RF；指标并行worker。
\end{enumerate}

\subsection{需要什么样的数据}
\begin{itemize}
  \item 基线 + HbA1c日序列 + 死亡/出院时间 + 复合指标日序列。
  \item 公式含糖尿病/HbA1c的指标不入队；现行设计不排除基线已有糖尿病。
\end{itemize}

\subsection{典型流水线}
\begin{center}
\small
\texttt{指标暴露 $\rightarrow$ 排除/trim $\rightarrow$ 轨迹}\\
\texttt{$\rightarrow$ Fine-Gray / CIF / RCS / 敏感性 / pub}
\end{center}
